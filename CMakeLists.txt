cmake_minimum_required(VERSION 3.16)
project(DinnDex VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED)
find_package(exiv2 CONFIG REQUIRED)
find_package(nlohmann_json 3.2.0 REQUIRED)
add_library(dinndex_core cpp/src/scanner.cpp cpp/src/hashing.cpp cpp/src/metadata.cpp)
target_include_directories(dinndex_core PUBLIC cpp/include)
target_link_libraries(dinndex_core PUBLIC OpenSSL::Crypto exiv2lib nlohmann_json::nlohmann_json)
target_compile_options(dinndex_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dinndex cpp/src/main.cpp)
target_link_libraries(dinndex PRIVATE dinndex_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests cpp/tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE dinndex_core)
  add_test(NAME core COMMAND core_tests)
endif()
